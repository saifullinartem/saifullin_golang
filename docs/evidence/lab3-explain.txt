PREPARE stale_trip_batch(interval,int,timestamptz,uuid) AS
SELECT id, user_id, driver_id, start_latitude, start_longitude, end_latitude, end_longitude, price, status, started_at, finished_at, last_position_at, COALESCE(last_position_at, started_at) AS stale_at
		FROM trips
		WHERE status = 'active' AND COALESCE(last_position_at, started_at) < now() - $1::interval
	 AND (COALESCE(last_position_at, started_at), id) > ($3::timestamptz, $4::uuid) ORDER BY COALESCE(last_position_at, started_at), id LIMIT $2;
SELECT count(*) AS trips FROM trips;
